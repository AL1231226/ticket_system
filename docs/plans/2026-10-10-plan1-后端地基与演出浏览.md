```java
package com.liule.ticket.config;

import cn.hutool.core.util.StrUtil;
import org.redisson.Redisson;
import org.redisson.api.RedissonClient;
import org.redisson.config.Config;
import org.springframework.beans.factory.annotation.Value;
import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

/**
 * Redisson 单机客户端。
 * <p>
 * 原版声明了 redissonClient / redissonClient1 / redissonClient2 三个 Bean 且端口硬编码，
 * 那是学集群锁时的临时写法，这里收敛成一个，地址从 application.yaml 读。
 */
@Configuration
public class RedissonConfig {

    @Value("${spring.data.redis.host}")
    private String host;

    @Value("${spring.data.redis.port}")
    private int port;

    @Value("${spring.data.redis.password:}")
    private String password;

    @Bean(destroyMethod = "shutdown")
    public RedissonClient redissonClient() {
        Config config = new Config();
        config.useSingleServer()
                .setAddress("redis://" + host + ":" + port)
                .setPassword(StrUtil.isBlank(password) ? null : password);
        return Redisson.create(config);
    }
}
```

> `destroyMethod = "shutdown"` 不写的话，应用关闭时 Redisson 的 Netty 线程池还在跑，进程会卡住不退出。

```java
package com.liule.ticket.config;

import org.springframework.context.annotation.Bean;
import org.springframework.context.annotation.Configuration;

import java.util.concurrent.LinkedBlockingQueue;
import java.util.concurrent.ThreadPoolExecutor;
import java.util.concurrent.TimeUnit;
import java.util.concurrent.atomic.AtomicInteger;

/**
 * 缓存重建线程池。
 * <p>
 * 只给"逻辑过期重建"用。原版用 Executors.newFixedThreadPool(10) 的静态字段，
 * 问题有两个：一是无界队列会堆积任务直到 OOM，二是静态字段不受 Spring 管理、无法优雅关闭。
 */
@Configuration
public class ThreadPoolConfig {

    private static final AtomicInteger POOL_SEQ = new AtomicInteger(1);

    @Bean(name = "cacheRebuildExecutor")
    public ThreadPoolExecutor cacheRebuildExecutor() {
        int coreSize = Runtime.getRuntime().availableProcessors();
        return new ThreadPoolExecutor(
                coreSize,                       // 核心线程数
                coreSize * 2,                   // 最大线程数
                60L, TimeUnit.SECONDS,          // 空闲回收时间
                new LinkedBlockingQueue<>(200), // 有界队列，满了触发拒绝策略
                r -> new Thread(r, "cache-rebuild-" + POOL_SEQ.getAndIncrement()),
                // 队列满了以后由调用线程自己执行：宁可拖慢这一个请求，也不丢重建任务
                new ThreadPoolExecutor.CallerRunsPolicy()
        );
    }
}
```

```java
package com.liule.ticket.utils;

import cn.hutool.core.util.StrUtil;

/**
 * 正则校验。方法名故意用 isXxxInvalid —— 调用处写成 if (isPhoneInvalid(phone)) 语义更直白。
 */
public final class RegexUtils {

    private RegexUtils() {
    }

    public static boolean isPhoneInvalid(String phone) {
        return mismatch(phone, RegexPatterns.PHONE_REGEX);
    }

    public static boolean isEmailInvalid(String email) {
        return mismatch(email, RegexPatterns.EMAIL_REGEX);
    }

    public static boolean isCodeInvalid(String code) {
        return mismatch(code, RegexPatterns.VERIFY_CODE_REGEX);
    }

    public static boolean isPasswordInvalid(String password) {
        return mismatch(password, RegexPatterns.PASSWORD_REGEX);
    }

    private static boolean mismatch(String str, String regex) {
        if (StrUtil.isBlank(str)) {
            return true;
        }
        return !str.matches(regex);
    }
}
```

```java
package com.liule.ticket.utils;

import cn.hutool.core.util.RandomUtil;
import org.springframework.util.DigestUtils;

import java.nio.charset.StandardCharsets;

/**
 * 加盐 MD5。存储格式：{salt}@{md5(password + salt)}
 * <p>
 * 生产环境应该用 BCrypt，这里保持与原项目一致，方便对照讲解。
 */
public final class PasswordEncoder {

    private PasswordEncoder() {
    }

    private static final String SALT_SEPARATOR = "@";

    public static String encode(String password) {
        String salt = RandomUtil.randomString(20);
        return encode(password, salt);
    }

    private static String encode(String password, String salt) {
        String digest = DigestUtils.md5DigestAsHex(
                (password + salt).getBytes(StandardCharsets.UTF_8));
        return salt + SALT_SEPARATOR + digest;
    }

    public static Boolean matches(String encodedPassword, String rawPassword) {
        if (encodedPassword == null || rawPassword == null) {
            return false;
        }
        if (!encodedPassword.contains(SALT_SEPARATOR)) {
            throw new IllegalArgumentException("密码格式不正确");
        }
        String salt = encodedPassword.split(SALT_SEPARATOR)[0];
        return encodedPassword.equals(encode(rawPassword, salt));
    }
}
```

```java
package com.liule.ticket.service.impl;

import cn.hutool.core.bean.BeanUtil;
import cn.hutool.core.lang.UUID;
import cn.hutool.core.util.RandomUtil;
import cn.hutool.core.util.StrUtil;
import com.baomidou.mybatisplus.extension.service.impl.ServiceImpl;
import com.liule.ticket.common.RedisConstants;
import com.liule.ticket.common.Result;
import com.liule.ticket.common.SystemConstants;
import com.liule.ticket.common.UserHolder;
import com.liule.ticket.dto.LoginFormDTO;
import com.liule.ticket.dto.UserDTO;
import com.liule.ticket.entity.User;
import com.liule.ticket.entity.UserInfo;
import com.liule.ticket.mapper.UserMapper;
import com.liule.ticket.service.IUserInfoService;
import com.liule.ticket.service.IUserService;
import com.liule.ticket.utils.RegexUtils;
import lombok.RequiredArgsConstructor;
import lombok.extern.slf4j.Slf4j;
import org.springframework.data.redis.connection.BitFieldSubCommands;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Service;
import org.springframework.transaction.annotation.Transactional;

import java.time.LocalDateTime;
import java.time.format.DateTimeFormatter;
import java.util.HashMap;
import java.util.List;
import java.util.Map;
import java.util.concurrent.TimeUnit;

@Slf4j
@Service
@RequiredArgsConstructor
public class UserServiceImpl extends ServiceImpl<UserMapper, User> implements IUserService {

    private final StringRedisTemplate stringRedisTemplate;
    private final IUserInfoService userInfoService;

    private static final DateTimeFormatter SIGN_MONTH_FORMAT = DateTimeFormatter.ofPattern("yyyyMM");

    @Override
    public Result sendCode(String phone) {
        if (RegexUtils.isPhoneInvalid(phone)) {
            return Result.fail("手机号格式错误");
        }
        String code = RandomUtil.randomNumbers(6);
        stringRedisTemplate.opsForValue()
                .set(RedisConstants.LOGIN_CODE_KEY + phone, code,
                        RedisConstants.LOGIN_CODE_TTL, TimeUnit.MINUTES);
        // 真实项目这里调短信服务商；写日志是为了本地开发能拿到验证码
        log.debug("【模拟短信】手机号 {} 的验证码是 {}", phone, code);
        return Result.ok();
    }

    @Override
    @Transactional(rollbackFor = Exception.class)
    public Result login(LoginFormDTO loginForm) {
        String phone = loginForm.getPhone();
        if (RegexUtils.isPhoneInvalid(phone)) {
            return Result.fail("手机号格式错误");
        }

        String cacheCode = stringRedisTemplate.opsForValue().get(RedisConstants.LOGIN_CODE_KEY + phone);
        if (cacheCode == null || !cacheCode.equals(loginForm.getCode())) {
            return Result.fail("验证码错误");
        }
        // 验证码一次性使用，验证通过立刻删除，防止重放
        stringRedisTemplate.delete(RedisConstants.LOGIN_CODE_KEY + phone);

        User user = query().eq("phone", phone).one();
        if (user == null) {
            user = registerByPhone(phone);
        }
        return Result.ok(createToken(user));
    }

    @Override
    public Result logout(String token) {
        if (StrUtil.isNotBlank(token)) {
            stringRedisTemplate.delete(RedisConstants.LOGIN_USER_KEY + token);
        }
        return Result.ok();
    }

    @Override
    public Result sign() {
        Long userId = UserHolder.getUser().getId();
        LocalDateTime now = LocalDateTime.now();
        String key = RedisConstants.SIGN_KEY + userId + ":" + now.format(SIGN_MONTH_FORMAT);
        // 第 dayOfMonth 天，就把第 (dayOfMonth - 1) 位置 1
        stringRedisTemplate.opsForValue().setBit(key, now.getDayOfMonth() - 1L, true);
        return Result.ok();
    }

    @Override
    public Result countContinuousSign() {
        Long userId = UserHolder.getUser().getId();
        LocalDateTime now = LocalDateTime.now();
        String key = RedisConstants.SIGN_KEY + userId + ":" + now.format(SIGN_MONTH_FORMAT);
        int dayOfMonth = now.getDayOfMonth();

        // 取出本月 1 号到今天的所有 bit，拼成一个整数
        List<Long> bits = stringRedisTemplate.opsForValue().bitField(key,
                BitFieldSubCommands.create()
                        .get(BitFieldSubCommands.BitFieldType.unsigned(dayOfMonth))
                        .valueAt(0));
        if (bits == null || bits.isEmpty() || bits.get(0) == null) {
            return Result.ok(0);
        }

        long num = bits.get(0);
        int count = 0;
        // 从最低位（今天）往前数连续的 1
        while (num != 0 && (num & 1L) == 1L) {
            count++;
            num >>>= 1;
        }
        return Result.ok(count);
    }

    @Override
    public Result queryPublicProfile(Long userId) {
        User user = getById(userId);
        if (user == null) {
            return Result.fail("用户不存在");
        }
        UserDTO dto = BeanUtil.copyProperties(user, UserDTO.class);
        UserInfo info = userInfoService.getById(userId);
        if (info != null) {
            info.setCreateTime(null);
            info.setUpdateTime(null);
        }
        Map<String, Object> data = new HashMap<>(2);
        data.put("user", dto);
        data.put("info", info);
        return Result.ok(data);
    }

    /** 手机号首次登录即注册 */
    private User registerByPhone(String phone) {
        User user = new User()
                .setPhone(phone)
                .setNickname(SystemConstants.USER_NICKNAME_PREFIX + RandomUtil.randomString(8))
                .setIcon("https://picsum.photos/seed/u" + phone + "/200/200");
        save(user);
        // 用户详情也要初始化一条，否则详情页拿不到数据
        userInfoService.save(new UserInfo().setUserId(user.getId()));
        log.debug("新用户注册成功，id={}, phone={}", user.getId(), phone);
        return user;
    }

    /** 生成 token，并把 UserDTO 以 Hash 结构写入 Redis（每个字段单独存，方便局部更新） */
    private String createToken(User user) {
        String token = UUID.randomUUID().toString(true);

        UserDTO userDTO = BeanUtil.copyProperties(user, UserDTO.class);
        Map<String, Object> beanMap = BeanUtil.beanToMap(userDTO);
        Map<String, String> userMap = new HashMap<>(beanMap.size());
        // 原版直接 v.toString()，字段为 null 时会 NPE；这里跳过 null
        beanMap.forEach((k, v) -> userMap.put(k, v == null ? "" : v.toString()));

        stringRedisTemplate.opsForHash().putAll(RedisConstants.LOGIN_USER_KEY + token, userMap);
        stringRedisTemplate.expire(RedisConstants.LOGIN_USER_KEY + token,
                RedisConstants.LOGIN_USER_TTL, TimeUnit.MINUTES);
        return token;
    }
}
```

`src/main/java/com/liule/ticket/interceptor/RefreshTokenInterceptor.java`

```java
package com.liule.ticket.interceptor;

import cn.hutool.core.bean.BeanUtil;
import cn.hutool.core.util.StrUtil;
import com.liule.ticket.common.RedisConstants;
import com.liule.ticket.common.UserHolder;
import com.liule.ticket.dto.UserDTO;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.web.servlet.HandlerInterceptor;

import java.util.Map;
import java.util.concurrent.TimeUnit;

/**
 * 第一道拦截器：把 token 换成 UserDTO 放进 ThreadLocal，并给 token 续期。
 * <p>
 * 永远返回 true —— 它的职责是"尝试登录"，不是"拦截"。真正的拦截交给 LoginInterceptor。
 * <p>
 * 这样拆分的好处：登录接口也能拿到当前用户（如果需要），且"续期"和"鉴权"两件事各自独立。
 */
public class RefreshTokenInterceptor implements HandlerInterceptor {

    private final StringRedisTemplate stringRedisTemplate;

    public RefreshTokenInterceptor(StringRedisTemplate stringRedisTemplate) {
        this.stringRedisTemplate = stringRedisTemplate;
    }

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        String token = request.getHeader("authorization");
        if (StrUtil.isBlank(token)) {
            return true;
        }

        Map<Object, Object> userMap = stringRedisTemplate.opsForHash()
                .entries(RedisConstants.LOGIN_USER_KEY + token);
        // 原版写的是 userMap == null，但 Redis 的 entries() 永远返回空 Map 而不是 null，
        // 所以那个判断从来没生效过，等于没做登录校验 —— 这里改成 isEmpty()
        if (userMap.isEmpty()) {
            return true;
        }

        UserHolder.saveUser(BeanUtil.fillBeanWithMap(userMap, new UserDTO(), false));
        // 滑动过期：只要还在用，就不用重新登录
        stringRedisTemplate.expire(RedisConstants.LOGIN_USER_KEY + token,
                RedisConstants.LOGIN_USER_TTL, TimeUnit.MINUTES);
        return true;
    }

    @Override
    public void afterCompletion(HttpServletRequest request, HttpServletResponse response,
                                Object handler, Exception ex) {
        // 必须清理：Tomcat 线程是复用的，不清理会把上一个请求的用户带给下一个请求
        UserHolder.removeUser();
    }
}
```

```java
package com.liule.ticket.interceptor;

import com.liule.ticket.common.UserHolder;
import jakarta.servlet.http.HttpServletRequest;
import jakarta.servlet.http.HttpServletResponse;
import org.springframework.web.servlet.HandlerInterceptor;

/**
 * 第二道拦截器：只看 ThreadLocal 里有没有用户。
 * 没有就返回 401，不再往下走。
 */
public class LoginInterceptor implements HandlerInterceptor {

    @Override
    public boolean preHandle(HttpServletRequest request, HttpServletResponse response, Object handler) {
        if (UserHolder.getUser() == null) {
            response.setStatus(HttpServletResponse.SC_UNAUTHORIZED);
            return false;
        }
        return true;
    }
}
```

> `LoginInterceptor` **不要**再写 `afterCompletion` 去 `removeUser()`。清理只在 `RefreshTokenInterceptor` 里做一次就行 —— 两个拦截器都清，`afterCompletion` 是按注册顺序倒着执行的（Login 先清、Refresh 后清），看起来没问题，但一旦调整 order 顺序就会出现"清早了"。单一职责更安全。

```java
package com.liule.ticket.config;

import com.liule.ticket.interceptor.LoginInterceptor;
import com.liule.ticket.interceptor.RefreshTokenInterceptor;
import jakarta.annotation.Resource;
import org.springframework.context.annotation.Configuration;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.web.servlet.config.annotation.InterceptorRegistry;
import org.springframework.web.servlet.config.annotation.WebMvcConfigurer;

@Configuration
public class WebMvcConfig implements WebMvcConfigurer {

    @Resource
    private StringRedisTemplate stringRedisTemplate;

    @Override
    public void addInterceptors(InterceptorRegistry registry) {
        // order 小的先执行
        registry.addInterceptor(new RefreshTokenInterceptor(stringRedisTemplate))
                .addPathPatterns("/**")
                .order(0);

        registry.addInterceptor(new LoginInterceptor())
                .excludePathPatterns(
                        "/user/code",       // 发验证码
                        "/user/login",      // 登录
                        "/show/*",          // 演出详情（只匹配一段，不会命中 /show/1/want）
                        "/show/categories", // 分类列表
                        "/show/list",       // 演出列表
                        "/venue/*",         // 场馆详情
                        "/error"            // Spring Boot 默认错误页
                )
                .order(1);
    }
}
```

> **为什么用 `/show/*` 而不是 `/show/**`**：`/show/**` 会把 `POST /show/{id}/want`（标记想看，需要登录）一起放行，业务代码里 `UserHolder.getUser()` 就是 null，直接 NPE。`/show/*` 只匹配单段路径，`/show/1` 放行、`/show/1/want` 仍然拦截。
>
> **为什么不写 `/user/{id}`**：Spring 的路径匹配里 `{id}` 能匹配任意单段，`/user/me`、`/user/sign` 都会被它匹配到从而被放行，`POST /user/sign` 就会因为拿不到用户而 500。所以用户主页的路径故意设计成 `/user/profile/{id}`，避开这个坑。

```java
import java.util.concurrent.ExecutorService;
import java.util.concurrent.Executors;
import java.util.concurrent.TimeUnit;

import static org.assertj.core.api.Assertions.assertThat;

class RedisIdWorkerTest {

    @Autowired
    private RedisIdWorker redisIdWorker;

    @Test
    void idIncreasesMonotonically() {
        long first = redisIdWorker.nextId("test");
        long second = redisIdWorker.nextId("test");
        assertThat(second).isGreaterThan(first);
    }

    @Test
    void idIsUniqueUnderConcurrency() throws InterruptedException {
        int threadCount = 100;
        int perThread = 100;

        Set<Long> ids = java.util.Collections.synchronizedSet(new HashSet<>());
        CountDownLatch ready = new CountDownLatch(threadCount);
        CountDownLatch start = new CountDownLatch(1);
        CountDownLatch done = new CountDownLatch(threadCount);

        ExecutorService pool = Executors.newFixedThreadPool(threadCount);
        for (int i = 0; i < threadCount; i++) {
            pool.submit(() -> {
                ready.countDown();
                try {
                    start.await();               // 等所有线程就位，一起发车，制造真实竞争
                    for (int j = 0; j < perThread; j++) {
                        ids.add(redisIdWorker.nextId("test"));
                    }
                } catch (InterruptedException e) {
                    Thread.currentThread().interrupt();
                } finally {
                    done.countDown();
                }
            });
        }

        ready.await();
        start.countDown();
        done.await(30, TimeUnit.SECONDS);
        pool.shutdown();

        assertThat(ids).hasSize(threadCount * perThread);
    }
}
```

```java
package com.liule.ticket.utils;

import com.liule.ticket.common.RedisConstants;
import lombok.RequiredArgsConstructor;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import java.time.LocalDate;
import java.time.LocalDateTime;
import java.time.ZoneOffset;
import java.time.format.DateTimeFormatter;

/**
 * 全局唯一 ID：64 位 = 32 位时间戳（秒） + 32 位自增序列。
 * <pre>
 *  0 | 0000000 0000000 0000000 0000000 | 00000000 00000000 00000000 00000000
 * 符号位        31 位时间戳（秒）                32 位序列号
 * </pre>
 * 时间戳是"当前秒 - 起始秒"，所以够用约 68 年；
 * 序列号来自 Redis 的 INCR，按 {keyPrefix}:{yyyy:MM:dd} 分 key，每天清零，
 * 这样 key 的数量可控，也不会因为单个 key 无限增长而撑爆。
 */
@Component
@RequiredArgsConstructor
public class RedisIdWorker {

    /** 起始时间戳：2024-01-01 00:00:00 UTC（秒） */
    private static final long BEGIN_TIMESTAMP = 1704067200L;

    /** 序列号占用的位数 */
    private static final int COUNT_BITS = 32;

    private static final DateTimeFormatter DATE_FORMAT = DateTimeFormatter.ofPattern("yyyy:MM:dd");

    private final StringRedisTemplate stringRedisTemplate;

    public long nextId(String keyPrefix) {
        LocalDateTime now = LocalDateTime.now();
        // 时间戳和日期必须来自同一个时钟，否则跨零点时会算出对不上的 ID
        long timestamp = now.toEpochSecond(ZoneOffset.UTC) - BEGIN_TIMESTAMP;
        long sequence = stringRedisTemplate.opsForValue()
                .increment(RedisConstants.ID_WORKER_KEY + keyPrefix + ":" + now.format(DATE_FORMAT));
        return timestamp << COUNT_BITS | sequence;
    }
}
```

`src/main/java/com/liule/ticket/utils/CacheClient.java`

```java
package com.liule.ticket.utils;

import cn.hutool.core.util.BooleanUtil;
import cn.hutool.core.util.StrUtil;
import cn.hutool.json.JSONUtil;
import com.liule.ticket.common.RedisData;
import lombok.extern.slf4j.Slf4j;
import org.springframework.beans.factory.annotation.Qualifier;
import org.springframework.data.redis.core.StringRedisTemplate;
import org.springframework.stereotype.Component;

import java.time.Duration;
import java.time.LocalDateTime;
import java.util.concurrent.ThreadLocalRandom;
import java.util.concurrent.ThreadPoolExecutor;
import java.util.function.Supplier;

/**
 * 缓存工具：只负责 Redis 读写与缓存策略，不认识任何业务实体。
 * 回源逻辑由调用方以 {@link Supplier} 传进来。
 * <p>
 * 对应原版的 RedisClient，改动如下：
 * 1. 拼音方法名 → 英文（queryById_chuntou → queryWithLogicalExpire 等）
 * 2. 锁 key 前缀不再写死 LOCK_SHOP_KEY，改为参数传入
 * 3. 线程池从静态字段改为注入 Spring 管理的 Bean
 * 4. 补上 TTL 随机抖动（防雪崩），原版完全没做
 */
@Slf4j
@Component
public class CacheClient {

    /** 雪崩防护：TTL 抖动幅度（秒），实际 TTL 落在 [ttl-300, ttl+300] */
    private static final long TTL_JITTER_SECONDS = 300L;

    private final StringRedisTemplate stringRedisTemplate;
    private final ThreadPoolExecutor cacheRebuildExecutor;

    public CacheClient(StringRedisTemplate stringRedisTemplate,
                       @Qualifier("cacheRebuildExecutor") ThreadPoolExecutor cacheRebuildExecutor) {
        this.stringRedisTemplate = stringRedisTemplate;
        this.cacheRebuildExecutor = cacheRebuildExecutor;
    }

    /**
     * 写普通 JSON 缓存，TTL 带随机抖动。
     * 抖动是为了防雪崩：如果一批 key 同时写入、TTL 又完全相同，
     * 它们会在同一秒集体失效，请求洪峰直接砸到数据库。
     */
    public void set(String key, Object value, Duration ttl) {
        stringRedisTemplate.opsForValue()
                .set(key, JSONUtil.toJsonStr(value), jitter(ttl));
    }

    /**
     * 写逻辑过期缓存：数据不过期，过期时间存在 value 内部。
     */
    public void setWithLogicalExpire(String key, Object value, Duration ttl) {
        RedisData redisData = new RedisData();
        redisData.setData(value);
        redisData.setExpireTime(LocalDateTime.now().plus(jitter(ttl)));
        stringRedisTemplate.opsForValue().set(key, JSONUtil.toJsonStr(redisData));
    }

    /**
     * 读逻辑过期缓存。
     * <ul>
     *   <li>key 不存在 → 返回 {@code null}（由调用方决定是否回源、是否写空值缓存）</li>
     *   <li>未过期 → 直接返回</li>
     *   <li>已过期 → 抢到锁的线程提交异步重建任务，<b>所有线程都立刻拿到旧值返回</b></li>
     * </ul>
     * 与"互斥锁"方案的区别：互斥锁是让没抢到锁的线程等待，保证数据新鲜但会阻塞；
     * 逻辑过期是永远不阻塞，代价是短时间内读到旧数据。票务详情页读多写少、能容忍秒级旧数据，选逻辑过期。
     *
     * @param key        缓存 key（完整 key，含前缀）
     * @param lockKey    重建时的互斥锁 key
     * @param type       返回类型
     * @param dbFallback 缓存失效时的回源逻辑
     * @param ttl        逻辑过期时长（实际会加抖动）
     */
    public <R> R queryWithLogicalExpire(String key, String lockKey, Class<R> type,
                                        Supplier<R> dbFallback, Duration ttl) {
        String json = stringRedisTemplate.opsForValue().get(key);
        if (StrUtil.isBlank(json)) {
            return null;
        }

        RedisData redisData = JSONUtil.toBean(json, RedisData.class);
        R value = cn.hutool.core.bean.BeanUtil.toBean(redisData.getData(), type);

        if (redisData.getExpireTime().isAfter(LocalDateTime.now())) {
            return value;
        }

        // 已逻辑过期：只有抢到锁的线程去重建，其余线程拿着旧值直接返回
        if (tryLock(lockKey)) {
            cacheRebuildExecutor.submit(() -> rebuild(key, lockKey, type, dbFallback, ttl));
        }
        return value;
    }

    /** 异步重建。整个过程跑在线程池里，不能抛异常出去，否则线程池会静默吞掉。 */
    private <R> void rebuild(String key, String lockKey, Class<R> type,
                             Supplier<R> dbFallback, Duration ttl) {
        try {
            // 双重检查：从"读到过期"到"抢到锁"之间，可能已经有别的线程重建完了
            String latest = stringRedisTemplate.opsForValue().get(key);
            if (StrUtil.isNotBlank(latest)) {
                RedisData latestData = JSONUtil.toBean(latest, RedisData.class);
                if (latestData.getExpireTime().isAfter(LocalDateTime.now())) {
                    return;
                }
            }

            R fresh = dbFallback.get();
            if (fresh == null) {
                // 数据被删了：清掉缓存并写空值标记，交给调用方的空值缓存策略接管
                stringRedisTemplate.delete(key);
                log.warn("缓存重建时数据已不存在，key={}", key);
                return;
            }
            setWithLogicalExpire(key, fresh, ttl);
        } catch (Exception e) {
            log.error("缓存重建失败，key={}", key, e);
        } finally {
            // 注意：这里删锁有个经典缺陷 —— 如果重建耗时超过锁的 TTL，
            // 锁早已自动释放并被别的线程重新持有，此时删的是别人的锁。
            // Plan 2 引入 Redisson 后用它解决（持有者校验）。
            stringRedisTemplate.delete(lockKey);
        }
    }

    private boolean tryLock(String key) {
        Boolean flag = stringRedisTemplate.opsForValue()
                .setIfAbsent(key, "1", com.liule.ticket.common.RedisConstants.LOCK_SHOW_TTL,
                        java.util.concurrent.TimeUnit.SECONDS);
        return BooleanUtil.isTrue(flag);
    }

    /** 给 TTL 加 ±300 秒的随机抖动 */
    private Duration jitter(Duration ttl) {
        return ttl.plusSeconds(ThreadLocalRandom.current().nextLong(-TTL_JITTER_SECONDS, TTL_JITTER_SECONDS));
    }
}
```

```java
@Slf4j
@Service
@RequiredArgsConstructor
public class ShowServiceImpl extends ServiceImpl<ShowMapper, Show> implements IShowService {

    private final StringRedisTemplate stringRedisTemplate;
    private final CacheClient cacheClient;
    private final IShowSessionService showSessionService;
    private final ITicketTierService ticketTierService;

    @Override
    public Result queryDetailById(Long id) {
        String cacheKey = RedisConstants.CACHE_SHOW_KEY + id;
        String nullKey = RedisConstants.CACHE_NULL_SHOW_KEY + id;

        // 第一层：空值缓存（防穿透）。
        // 查一个不存在的演出 id，如果不做这层，每次请求都会打到数据库。
        if (Boolean.TRUE.equals(stringRedisTemplate.hasKey(nullKey))) {
            return Result.fail("演出不存在");
        }

        // 第二层：逻辑过期缓存（防击穿）。
        ShowDetailVO detail = cacheClient.queryWithLogicalExpire(
                cacheKey,
                RedisConstants.LOCK_SHOW_KEY + id,
                ShowDetailVO.class,
                () -> loadFromDb(id),
                Duration.ofMinutes(RedisConstants.CACHE_SHOW_TTL));

        if (detail == null) {
            // 第三层：缓存里没有（首次访问，或还没预热）
            detail = loadFromDb(id);
            if (detail == null) {
                stringRedisTemplate.opsForValue().set(nullKey, "",
                        RedisConstants.CACHE_NULL_TTL, TimeUnit.MINUTES);
                return Result.fail("演出不存在");
            }
            cacheClient.setWithLogicalExpire(cacheKey, detail,
                    Duration.ofMinutes(RedisConstants.CACHE_SHOW_TTL));
        }

        return Result.ok(detail);
    }

    /**
     * 从数据库装配详情。
     * 返回 null 表示演出不存在或已下架 —— 调用方据此写空值缓存。
     */
    private ShowDetailVO loadFromDb(Long id) {
        Show show = getById(id);
        if (show == null || Integer.valueOf(0).equals(show.getStatus())) {
            return null;
        }
        List<ShowSession> sessions = showSessionService.listWithTiersByShowId(id);
        show.setMinPrice(ticketTierService.lowestPriceByShowId(id));
        return new ShowDetailVO(show, sessions);
    }

}
```

```java
    @Override
    public Result queryShowList(String city, Long categoryId, String keyword, Integer page) {
        Page<Show> p = new Page<>(page == null || page < 1 ? 1 : page, SystemConstants.DEFAULT_PAGE_SIZE);

        LambdaQueryWrapper<Show> wrapper = new LambdaQueryWrapper<Show>()
                .eq(Show::getStatus, 1)
                .eq(StrUtil.isNotBlank(city), Show::getCity, city)
                .eq(categoryId != null, Show::getCategoryId, categoryId)
                .and(StrUtil.isNotBlank(keyword),
                        w -> w.like(Show::getName, keyword).or().like(Show::getArtist, keyword))
                .orderByAsc(Show::getStartDate);

        page(p, wrapper);

        List<Show> records = p.getRecords();
        // 列表页要显示"¥xxx 起"，逐条查最低价。演出数量在可接受范围内，
        // 如果以后列表变大，改成一次 GROUP BY 查回来。
        records.forEach(s -> s.setMinPrice(ticketTierService.lowestPriceByShowId(s.getId())));

        return Result.ok(records, p.getTotal());
    }

    @Override
    public Result preloadDetailCache() {
        List<Show> shows = query().eq("status", 1).list();
        if (shows.isEmpty()) {
            return Result.fail("没有可预热的演出");
        }
        int count = 0;
        for (Show show : shows) {
            ShowDetailVO detail = loadFromDb(show.getId());
            if (detail == null) {
                continue;
            }
            cacheClient.setWithLogicalExpire(
                    RedisConstants.CACHE_SHOW_KEY + show.getId(),
                    detail,
                    Duration.ofMinutes(RedisConstants.CACHE_SHOW_TTL));
            // 预热成功就把可能存在的空值缓存清掉，否则会把正确数据挡在外面
            stringRedisTemplate.delete(RedisConstants.CACHE_NULL_SHOW_KEY + show.getId());
            count++;
        }
        log.info("演出详情缓存预热完成，共 {} 条", count);
        return Result.ok(count);
    }
```

## 已知的取舍（面试会被问到，先记下来）

| 点 | Plan 1 的做法 | 更好的做法 / 什么时候改 |
|---|---|---|
| 删锁不校验持有者 | `finally` 里直接 `delete(lockKey)` | 重建耗时超过锁 TTL 时会误删别人的锁。Plan 2 换 Redisson |
| 预热的锁 key 复用 `LOCK_SHOW_TTL`（10s） | 够用 | 重建慢的业务要单独配锁 TTL |
| 列表页逐条查最低价 | 演出量小，可接受 | 演出上千条后改成一次 `GROUP BY show_id, MIN(price)` |
| 分类缓存不设过期 | 靠主动更新 | 需要在改分类的地方调 `redisTemplate.delete(CACHE_CATEGORY_KEY)` |
| 密码用加盐 MD5 | 与原项目一致 | 生产用 BCrypt（`spring-security-crypto` 已可用） |
| `minPrice` 在列表里逐条算 | N 次查询 | 同上，改批量 |
